import { detectClients, addManualClient, scanSharedFile } from './clientLocator';
import type { DetectedClient } from '@swg/contracts';

export type ProjectKind = 'client' | 'tre-set' | 'mod-project';

export function detectFolderKind(folderPath: string): ProjectKind {
  // If the folder contains a client exe + cfg .include chain → 'client'
  const cfgNames = ['swgemu.cfg', 'client.cfg'];
  for (const cfg of cfgNames) {
    if (fs.existsSync(path.join(folderPath, cfg))) {
      // Also check for at least one .tre file in folder or Live/ subdir
      const liveDir = path.join(folderPath, 'Live');
      const checkDir = fs.existsSync(liveDir) ? liveDir : folderPath;
      const hasTre = fs.readdirSync(checkDir).some(f => f.endsWith('.tre'));
      if (hasTre) return 'client';
    }
  }
  // Has .tre files but no cfg → 'tre-set'
  const hasTreRoot = fs.readdirSync(folderPath).some(f => f.endsWith('.tre'));
  if (hasTreRoot) return 'tre-set';
  return 'mod-project';
}

export interface ClientLayout {
  /** cfg filename (e.g. 'swgemu.cfg') */

  /** TRE dir relative to install root: 'Live' or '' (root) */
  treSubdir: string;
  /** maxSearchPriority from this release's chain (informational) */

const KNOWN_LAYOUTS: Array<{ release: string; layout: ClientLayout }> = [
  { release: 'SWG Infinity',   layout: { cfgFile: 'swgemu.cfg', treSubdir: 'Live', maxSearchPriority: 60 } },
  { release: 'SWGEmu',         layout: { cfgFile: 'swgemu.cfg', treSubdir: '',     maxSearchPriority: 27 } },
  // 'client.cfg' releases — cfgFile varies; treSubdir unknown; maintainer must override
];

export function resolveLayout(installPath: string): ClientLayout | null {
  for (const { layout } of KNOWN_LAYOUTS) {
    const cfgPath = path.join(installPath, layout.cfgFile);
    const treDir  = layout.treSubdir
      ? path.join(installPath, layout.treSubdir)
      : installPath;
    if (fs.existsSync(cfgPath) && fs.readdirSync(treDir).some(f => f.endsWith('.tre'))) {
      return layout;
    }
  }
  return null;  // caller surfaces manual-override UI (Surface 3)
}
